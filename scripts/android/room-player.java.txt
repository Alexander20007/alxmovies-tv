package __PKG__;

import android.graphics.Color;
import android.os.Handler;
import android.os.Looper;
import android.view.Gravity;
import android.view.TextureView;
import android.view.View;
import android.view.ViewGroup;
import android.view.WindowManager;
import android.webkit.WebView;
import android.widget.FrameLayout;
import androidx.media3.common.C;
import androidx.media3.common.Format;
import androidx.media3.common.MediaItem;
import androidx.media3.common.PlaybackException;
import androidx.media3.common.Player;
import androidx.media3.common.TrackSelectionOverride;
import androidx.media3.common.Tracks;
import androidx.media3.common.VideoSize;
import androidx.media3.common.text.CueGroup;
import androidx.media3.datasource.DefaultHttpDataSource;
import androidx.media3.exoplayer.ExoPlayer;
import androidx.media3.exoplayer.hls.HlsMediaSource;
import androidx.media3.ui.AspectRatioFrameLayout;
import androidx.media3.ui.SubtitleView;
import com.getcapacitor.JSArray;
import com.getcapacitor.JSObject;
import com.getcapacitor.Plugin;
import com.getcapacitor.PluginCall;
import com.getcapacitor.PluginMethod;
import com.getcapacitor.annotation.CapacitorPlugin;
import java.util.ArrayList;
import java.util.List;

/**
 * Player das salas: um ExoPlayer NATIVO desenhado ATRÁS da página (o WebView fica transparente nessa área). O controle da sala
 * (botões, chat flutuante, reações) continua sendo HTML por cima do vídeo. A página manda a posição do quadro de vídeo (setBounds)
 * e recebe os eventos (play, pause, seek, tempo, faixas, erro) pelo evento "room".
 */
@CapacitorPlugin(name = "AlxRoomPlayer")
public class AlxRoomPlayerPlugin extends Plugin {
  private ExoPlayer player;
  private FrameLayout host;      // camada cheia ATRÁS do WebView (aceita qualquer tipo de layout do pai)
  private FrameLayout container;    // quadro do vídeo, posicionado dentro de "host"
  private AspectRatioFrameLayout aspect;
  private TextureView texture;
  private SubtitleView subtitles;
  private final Handler ui = new Handler(Looper.getMainLooper());
  private final List<Object[]> videoTracks = new ArrayList<>(); // {Tracks.Group, Integer}
  private final List<Object[]> textTracks = new ArrayList<>();
  private boolean readySent = false;
  private boolean resumedPending = false;

  private final Runnable ticker = new Runnable() {
    @Override public void run() {
      if (player != null) {
        long d = player.getDuration();
        JSObject o = new JSObject();
        o.put("event", "time");
        o.put("timeMs", player.getCurrentPosition());
        o.put("durationMs", d == C.TIME_UNSET ? 0 : d);
        o.put("playing", player.getPlayWhenReady());
        notifyListeners("room", o);
        ui.postDelayed(this, 500);
      }
    }
  };

  private void emit(String event, long timeMs) {
    JSObject o = new JSObject();
    o.put("event", event);
    o.put("timeMs", timeMs);
    notifyListeners("room", o);
  }

  private WebView web() { return getBridge().getWebView(); }

  private void place(double x, double y, double w, double h, boolean visible) {
    if (container == null || host == null) return;
    try {
      float d = getContext().getResources().getDisplayMetrics().density;
      int[] wl = new int[2];
      int[] hl = new int[2];
      web().getLocationInWindow(wl);
      host.getLocationInWindow(hl);
      // "host" é um FrameLayout nosso, então estes parâmetros são sempre do tipo certo
      FrameLayout.LayoutParams lp = new FrameLayout.LayoutParams(Math.max(1, Math.round((float) w * d)), Math.max(1, Math.round((float) h * d)), Gravity.TOP | Gravity.START);
      lp.leftMargin = Math.round((float) x * d) + (wl[0] - hl[0]);
      lp.topMargin = Math.round((float) y * d) + (wl[1] - hl[1]);
      container.setLayoutParams(lp);
      container.setVisibility(visible ? View.VISIBLE : View.INVISIBLE);
    } catch (Exception ignored) {}
  }

  /** Cria a área de vídeo atrás do WebView e o ExoPlayer. */
  @PluginMethod
  public void open(final PluginCall call) {
    final double x = call.getDouble("x", 0d), y = call.getDouble("y", 0d), w = call.getDouble("w", 300d), h = call.getDouble("h", 170d);
    getActivity().runOnUiThread(() -> {
      try {
        if (container == null) {
          ViewGroup parent = (ViewGroup) web().getParent();
          container = new FrameLayout(getContext());
          container.setBackgroundColor(Color.BLACK);
          aspect = new AspectRatioFrameLayout(getContext());
          aspect.setResizeMode(AspectRatioFrameLayout.RESIZE_MODE_FIT);
          texture = new TextureView(getContext());
          aspect.addView(texture, new FrameLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, ViewGroup.LayoutParams.MATCH_PARENT));
          container.addView(aspect, new FrameLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, ViewGroup.LayoutParams.MATCH_PARENT, Gravity.CENTER));
          subtitles = new SubtitleView(getContext());
          container.addView(subtitles, new FrameLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, ViewGroup.LayoutParams.MATCH_PARENT));
          host = new FrameLayout(getContext());
          host.addView(container, new FrameLayout.LayoutParams(1, 1));
          // O pai do WebView no Capacitor é um CoordinatorLayout: parâmetros genéricos são convertidos pelo próprio Android.
          parent.addView(host, 0, new ViewGroup.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, ViewGroup.LayoutParams.MATCH_PARENT));
          web().setBackgroundColor(Color.TRANSPARENT); // a página enxerga o vídeo pelo "buraco" transparente
          getActivity().getWindow().addFlags(WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON);
        }
        place(x, y, w, h, true);
        if (player == null) {
          player = new ExoPlayer.Builder(getContext()).build();
          player.setVideoTextureView(texture);
          player.addListener(new Player.Listener() {
            @Override public void onPlaybackStateChanged(int state) {
              if (player == null) return;
              if (state == Player.STATE_BUFFERING) emit("waiting", player.getCurrentPosition());
              if (state == Player.STATE_READY) {
                if (!readySent) { readySent = true; emit("ready", player.getCurrentPosition()); }
                emit("playing", player.getCurrentPosition());
              }
              if (state == Player.STATE_ENDED) emit("ended", player.getCurrentPosition());
            }
            @Override public void onPlayWhenReadyChanged(boolean playWhenReady, int reason) {
              if (player == null || reason == Player.PLAY_WHEN_READY_CHANGE_REASON_END_OF_MEDIA_ITEM) return;
              emit(playWhenReady ? "play" : "pause", player.getCurrentPosition());
            }
            @Override public void onPositionDiscontinuity(Player.PositionInfo oldPos, Player.PositionInfo newPos, int reason) {
              if (reason == Player.DISCONTINUITY_REASON_SEEK) emit("seeking", newPos.positionMs);
            }
            @Override public void onVideoSizeChanged(VideoSize v) {
              if (v.height > 0 && aspect != null) aspect.setAspectRatio(v.width * v.pixelWidthHeightRatio / v.height);
            }
            @Override public void onCues(CueGroup g) { if (subtitles != null) subtitles.setCues(g.cues); }
            @Override public void onTracksChanged(Tracks t) { sendTracks(t); }
            @Override public void onPlayerError(PlaybackException e) {
              JSObject o = new JSObject();
              o.put("event", "error");
              o.put("message", e.getMessage() == null ? "Falha ao reproduzir o vídeo" : e.getMessage());
              o.put("timeMs", player == null ? 0 : player.getCurrentPosition());
              notifyListeners("room", o);
            }
          });
          ui.removeCallbacks(ticker);
          ui.post(ticker);
        }
        call.resolve();
      } catch (Exception e) {
        call.reject(e.getMessage() == null ? "Não foi possível abrir o player nativo" : e.getMessage());
      }
    });
  }

  /** Troca o vídeo (primeiro vídeo da sala, próximo episódio, próximo da fila). Não toca sozinho. */
  @PluginMethod
  public void load(final PluginCall call) {
    final String url = call.getString("url", "");
    final long startMs = call.getLong("startMs", 0L);
    getActivity().runOnUiThread(() -> {
      if (player == null || url.isEmpty()) { call.reject("Player não está aberto"); return; }
      DefaultHttpDataSource.Factory http = new DefaultHttpDataSource.Factory()
        .setAllowCrossProtocolRedirects(true)
        .setConnectTimeoutMs(15000)
        .setReadTimeoutMs(25000)
        .setUserAgent("Mozilla/5.0 (Linux; Android 13) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0 Mobile Safari/537.36");
      readySent = false;
      player.setPlayWhenReady(false);
      player.setMediaSource(new HlsMediaSource.Factory(http).createMediaSource(MediaItem.fromUri(url)));
      player.prepare();
      if (startMs > 0) player.seekTo(startMs);
      call.resolve();
    });
  }

  @PluginMethod
  public void setBounds(final PluginCall call) {
    final double x = call.getDouble("x", 0d), y = call.getDouble("y", 0d), w = call.getDouble("w", 300d), h = call.getDouble("h", 170d);
    final boolean visible = Boolean.TRUE.equals(call.getBoolean("visible", true));
    getActivity().runOnUiThread(() -> { place(x, y, w, h, visible); call.resolve(); });
  }

  @PluginMethod
  public void play(final PluginCall call) {
    getActivity().runOnUiThread(() -> { if (player != null) player.play(); call.resolve(); });
  }

  @PluginMethod
  public void pause(final PluginCall call) {
    getActivity().runOnUiThread(() -> { if (player != null) player.pause(); call.resolve(); });
  }

  @PluginMethod
  public void seekTo(final PluginCall call) {
    final long ms = call.getLong("ms", 0L);
    getActivity().runOnUiThread(() -> { if (player != null) player.seekTo(Math.max(0, ms)); call.resolve(); });
  }

  @PluginMethod
  public void setVolume(final PluginCall call) {
    final double v = call.getDouble("v", 100d);
    getActivity().runOnUiThread(() -> { if (player != null) player.setVolume((float) Math.max(0d, Math.min(1d, v / 100d))); call.resolve(); });
  }

  @PluginMethod
  public void setQuality(final PluginCall call) {
    final int index = call.getInt("index", -1);
    getActivity().runOnUiThread(() -> {
      if (player != null) {
        if (index < 0 || index >= videoTracks.size()) {
          player.setTrackSelectionParameters(player.getTrackSelectionParameters().buildUpon().clearOverridesOfType(C.TRACK_TYPE_VIDEO).build());
        } else {
          Tracks.Group g = (Tracks.Group) videoTracks.get(index)[0];
          int i = (Integer) videoTracks.get(index)[1];
          player.setTrackSelectionParameters(player.getTrackSelectionParameters().buildUpon()
            .setOverrideForType(new TrackSelectionOverride(g.getMediaTrackGroup(), i)).build());
        }
      }
      call.resolve();
    });
  }

  @PluginMethod
  public void setSubtitle(final PluginCall call) {
    final int index = call.getInt("index", -1);
    getActivity().runOnUiThread(() -> {
      if (player != null) {
        if (index < 0 || index >= textTracks.size()) {
          player.setTrackSelectionParameters(player.getTrackSelectionParameters().buildUpon().setTrackTypeDisabled(C.TRACK_TYPE_TEXT, true).build());
        } else {
          Tracks.Group g = (Tracks.Group) textTracks.get(index)[0];
          int i = (Integer) textTracks.get(index)[1];
          player.setTrackSelectionParameters(player.getTrackSelectionParameters().buildUpon()
            .setTrackTypeDisabled(C.TRACK_TYPE_TEXT, false)
            .setOverrideForType(new TrackSelectionOverride(g.getMediaTrackGroup(), i)).build());
        }
      }
      call.resolve();
    });
  }

  private void sendTracks(Tracks t) {
    videoTracks.clear();
    textTracks.clear();
    JSArray levels = new JSArray();
    JSArray subs = new JSArray();
    int quality = -1, sub = -1;
    boolean textOff = player != null && player.getTrackSelectionParameters().disabledTrackTypes.contains(C.TRACK_TYPE_TEXT);
    boolean videoOverride = false;
    if (player != null) {
      for (TrackSelectionOverride o : player.getTrackSelectionParameters().overrides.values()) if (o.getType() == C.TRACK_TYPE_VIDEO) videoOverride = true;
    }
    for (Tracks.Group g : t.getGroups()) {
      if (g.getType() == C.TRACK_TYPE_VIDEO) {
        for (int i = 0; i < g.length; i++) {
          if (!g.isTrackSupported(i)) continue;
          Format f = g.getTrackFormat(i);
          JSObject l = new JSObject();
          l.put("index", videoTracks.size());
          l.put("height", f.height > 0 ? f.height : 0);
          l.put("bitrate", f.bitrate > 0 ? f.bitrate : 0);
          levels.put(l);
          if (videoOverride && g.isTrackSelected(i)) quality = videoTracks.size();
          videoTracks.add(new Object[] { g, i });
        }
      } else if (g.getType() == C.TRACK_TYPE_TEXT) {
        for (int i = 0; i < g.length; i++) {
          Format f = g.getTrackFormat(i);
          JSObject s = new JSObject();
          s.put("index", textTracks.size());
          s.put("label", f.label != null ? f.label : (f.language != null ? f.language : "Legenda " + (textTracks.size() + 1)));
          subs.put(s);
          if (!textOff && g.isTrackSelected(i)) sub = textTracks.size();
          textTracks.add(new Object[] { g, i });
        }
      }
    }
    JSObject o = new JSObject();
    o.put("event", "tracks");
    o.put("levels", levels);
    o.put("subs", subs);
    o.put("quality", quality);
    o.put("sub", sub);
    notifyListeners("room", o);
  }

  @PluginMethod
  public void close(final PluginCall call) {
    getActivity().runOnUiThread(() -> { release(); call.resolve(); });
  }

  private void release() {
    ui.removeCallbacks(ticker);
    if (player != null) { try { player.release(); } catch (Exception ignored) {} player = null; }
    if (host != null) {
      try { ((ViewGroup) host.getParent()).removeView(host); } catch (Exception ignored) {}
    }
    if (container != null) {
      host = null; container = null; aspect = null; texture = null; subtitles = null;
      try { web().setBackgroundColor(Color.parseColor("#000000")); } catch (Exception ignored) {}
      try { getActivity().getWindow().clearFlags(WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON); } catch (Exception ignored) {}
    }
    videoTracks.clear();
    textTracks.clear();
    readySent = false;
  }

  // O app foi para o segundo plano: pausa sem avisar a sala (não pode pausar para todo mundo).
  @Override
  protected void handleOnStop() {
    super.handleOnStop();
    if (player != null) { resumedPending = true; player.setPlayWhenReady(false); }
  }

  @Override
  protected void handleOnResume() {
    super.handleOnResume();
    if (player != null && resumedPending) { resumedPending = false; emit("resumed", player.getCurrentPosition()); }
  }

  @Override
  protected void handleOnDestroy() {
    release();
    super.handleOnDestroy();
  }
}
