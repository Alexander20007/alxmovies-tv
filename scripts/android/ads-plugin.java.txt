package __PKG__;

import android.app.Activity;
import android.os.Handler;
import android.os.Looper;
import android.view.Gravity;
import android.view.View;
import android.view.ViewGroup;
import android.widget.FrameLayout;
import androidx.core.graphics.Insets;
import androidx.core.view.ViewCompat;
import androidx.core.view.WindowInsetsCompat;
import com.getcapacitor.JSObject;
import com.getcapacitor.Plugin;
import com.getcapacitor.PluginCall;
import com.getcapacitor.PluginMethod;
import com.getcapacitor.annotation.CapacitorPlugin;
import com.startapp.sdk.ads.banner.Banner;
import com.startapp.sdk.ads.banner.BannerListener;
import com.startapp.sdk.adsbase.Ad;
import com.startapp.sdk.adsbase.StartAppAd;
import com.startapp.sdk.adsbase.StartAppSDK;
import com.startapp.sdk.adsbase.adlisteners.AdDisplayListener;
import com.startapp.sdk.adsbase.adlisteners.AdEventListener;
import com.startapp.sdk.adsbase.adlisteners.VideoListener;
import com.unity3d.ads.IUnityAdsInitializationListener;
import com.unity3d.ads.IUnityAdsLoadListener;
import com.unity3d.ads.IUnityAdsShowListener;
import com.unity3d.ads.UnityAds;
import com.unity3d.ads.UnityAdsShowOptions;
import com.unity3d.services.banners.BannerErrorInfo;
import com.unity3d.services.banners.BannerView;
import com.unity3d.services.banners.UnityBannerSize;
import java.util.concurrent.atomic.AtomicBoolean;

/**
 * Anúncios com DUAS redes: Start.io e Unity Ads.
 * Regra (definida em src/lib/config.ts):
 *  - "fallback":  usa a rede principal; se ela não tiver anúncio pronto (ou falhar), usa a outra.
 *  - "alternate": reveza a rede a cada anúncio (metade para cada); se a da vez não tiver anúncio, usa a outra.
 * O JavaScript decide QUANDO mostrar (sorteio, intervalo, plano); aqui só carregamos e exibimos, sempre na thread principal.
 */
@CapacitorPlugin(name = "AlxAds")
public class AlxAdsPlugin extends Plugin {
  private static final int BANNER_W = 320, BANNER_H = 50;
  private static final String SIO = "startio", UNITY = "unity";

  private final Handler main = new Handler(Looper.getMainLooper());

  // configuração
  private String primary = SIO, strategy = "fallback";
  private boolean flip = false;
  private String uInter, uReward, uBanner;
  private boolean initStarted = false;

  // estado de cada rede
  private boolean sReady = false, uReady = false;
  private StartAppAd sInter, sReward;
  private boolean sInterOk = false, sRewardOk = false, sRewardEarned = false;
  private boolean uInterOk = false, uRewardOk = false;

  private View banner;

  // ------------------------------------------------------------------ init
  @PluginMethod
  public void init(final PluginCall call) {
    if (initStarted) { resolveInit(call); return; }
    initStarted = true;
    final String appId = call.getString("appId");
    final String gameId = call.getString("unityGameId");
    final boolean test = Boolean.TRUE.equals(call.getBoolean("testMode", false));
    uInter = call.getString("unityInterstitial");
    uReward = call.getString("unityRewarded");
    uBanner = call.getString("unityBanner");
    primary = UNITY.equals(call.getString("primary")) ? UNITY : SIO;
    strategy = "alternate".equals(call.getString("strategy")) ? "alternate" : "fallback";

    final boolean wantS = appId != null && !appId.isEmpty();
    final boolean wantU = gameId != null && !gameId.isEmpty();
    if (!wantS && !wantU) { call.reject("nenhuma rede configurada"); return; }

    final int[] pending = { (wantS ? 1 : 0) + (wantU ? 1 : 0) };
    final AtomicBoolean answered = new AtomicBoolean(false);
    final Runnable done = new Runnable() {
      @Override public void run() {
        synchronized (pending) { pending[0]--; if (pending[0] > 0) return; }
        if (answered.compareAndSet(false, true)) resolveInit(call);
      }
    };
    // se alguma rede não responder, não trava o app: responde com o que já estiver pronto
    main.postDelayed(new Runnable() { @Override public void run() { if (answered.compareAndSet(false, true)) resolveInit(call); } }, 9000);

    if (wantS) {
      try {
        StartAppSDK.setTestAdsEnabled(test);
        StartAppSDK.initParams(getContext().getApplicationContext(), appId)
          .setReturnAdsEnabled(false)            // sem anúncio ao voltar do segundo plano: quem decide é o sorteio do app
          .setCallback(new Runnable() {
            @Override public void run() { sReady = true; loadStartInter(); loadStartReward(); done.run(); }
          })
          .init();
      } catch (Throwable t) { done.run(); }
    }
    if (wantU) {
      try {
        UnityAds.initialize(getContext().getApplicationContext(), gameId, test, new IUnityAdsInitializationListener() {
          @Override public void onInitializationComplete() { uReady = true; loadUnityInter(); loadUnityReward(); done.run(); }
          @Override public void onInitializationFailed(UnityAds.UnityAdsInitializationError error, String message) { done.run(); }
        });
      } catch (Throwable t) { done.run(); }
    }
  }

  private void resolveInit(PluginCall call) {
    JSObject r = new JSObject();
    r.put("ok", sReady || uReady);
    r.put("startio", sReady);
    r.put("unity", uReady);
    call.resolve(r);
  }

  /** Ordem das redes para ESTE anúncio, conforme a regra. */
  private String[] order() {
    String other = SIO.equals(primary) ? UNITY : SIO;
    boolean primFirst = true;
    if ("alternate".equals(strategy)) { primFirst = !flip; flip = !flip; }
    return primFirst ? new String[] { primary, other } : new String[] { other, primary };
  }

  // ------------------------------------------------------------------ carregar
  private void loadStartInter() {
    final Activity act = getActivity();
    if (!sReady || act == null) return;
    sInterOk = false;
    final StartAppAd ad = new StartAppAd(act);
    sInter = ad;
    ad.loadAd(new AdEventListener() {
      @Override public void onReceiveAd(Ad a) { sInterOk = true; }
      @Override public void onFailedToReceiveAd(Ad a) { sInterOk = false; }
    });
  }

  private void loadStartReward() {
    final Activity act = getActivity();
    if (!sReady || act == null) return;
    sRewardOk = false; sRewardEarned = false;
    final StartAppAd ad = new StartAppAd(act);
    sReward = ad;
    ad.setVideoListener(new VideoListener() { @Override public void onVideoCompleted() { sRewardEarned = true; } });
    ad.loadAd(StartAppAd.AdMode.REWARDED_VIDEO, new AdEventListener() {
      @Override public void onReceiveAd(Ad a) { sRewardOk = true; }
      @Override public void onFailedToReceiveAd(Ad a) { sRewardOk = false; }
    });
  }

  private void loadUnityInter() {
    if (!uReady || uInter == null) return;
    uInterOk = false;
    UnityAds.load(uInter, new IUnityAdsLoadListener() {
      @Override public void onUnityAdsAdLoaded(String id) { uInterOk = true; }
      @Override public void onUnityAdsFailedToLoad(String id, UnityAds.UnityAdsLoadError e, String m) { uInterOk = false; }
    });
  }

  private void loadUnityReward() {
    if (!uReady || uReward == null) return;
    uRewardOk = false;
    UnityAds.load(uReward, new IUnityAdsLoadListener() {
      @Override public void onUnityAdsAdLoaded(String id) { uRewardOk = true; }
      @Override public void onUnityAdsFailedToLoad(String id, UnityAds.UnityAdsLoadError e, String m) { uRewardOk = false; }
    });
  }

  private void reloadInters() { loadStartInter(); loadUnityInter(); }
  private void reloadRewards() { loadStartReward(); loadUnityReward(); }

  // ------------------------------------------------------------------ intersticial
  @PluginMethod
  public void showInterstitial(final PluginCall call) {
    final Activity act = getActivity();
    if (act == null) { shownFalse(call); return; }
    for (final String prov : order()) {
      if (SIO.equals(prov) && sReady && sInterOk && sInter != null) {
        act.runOnUiThread(new Runnable() { @Override public void run() { showStartInter(call); } });
        return;
      }
      if (UNITY.equals(prov) && uReady && uInterOk) {
        act.runOnUiThread(new Runnable() { @Override public void run() { showUnityInter(act, call); } });
        return;
      }
    }
    reloadInters();           // nenhuma rede tinha anúncio pronto: tenta carregar para a próxima
    shownFalse(call);
  }

  private void shownFalse(PluginCall call) { JSObject r = new JSObject(); r.put("shown", false); call.resolve(r); }

  private void showStartInter(final PluginCall call) {
    sInterOk = false;
    final boolean[] done = { false }, shown = { false };
    sInter.showAd(new AdDisplayListener() {
      @Override public void adDisplayed(Ad a) { shown[0] = true; }
      @Override public void adClicked(Ad a) { }
      @Override public void adHidden(Ad a) { if (done[0]) return; done[0] = true; reloadInters(); resolveShown(call, true, SIO); }
      @Override public void adNotDisplayed(Ad a) { if (done[0]) return; done[0] = true; reloadInters(); resolveShown(call, shown[0], SIO); }
    });
  }

  private void showUnityInter(Activity act, final PluginCall call) {
    uInterOk = false;
    UnityAds.show(act, uInter, new UnityAdsShowOptions(), new IUnityAdsShowListener() {
      @Override public void onUnityAdsShowFailure(String id, UnityAds.UnityAdsShowError e, String m) { reloadInters(); resolveShown(call, false, UNITY); }
      @Override public void onUnityAdsShowStart(String id) { }
      @Override public void onUnityAdsShowClick(String id) { }
      @Override public void onUnityAdsShowComplete(String id, UnityAds.UnityAdsShowCompletionState s) { reloadInters(); resolveShown(call, true, UNITY); }
    });
  }

  private void resolveShown(PluginCall call, boolean shown, String provider) {
    JSObject r = new JSObject(); r.put("shown", shown); r.put("provider", provider); call.resolve(r);
  }

  // ------------------------------------------------------------------ premiado (só vale se o vídeo foi concluído)
  @PluginMethod
  public void showRewarded(final PluginCall call) {
    final Activity act = getActivity();
    if (act != null) {
      for (final String prov : order()) {
        if (SIO.equals(prov) && sReady && sRewardOk && sReward != null) {
          act.runOnUiThread(new Runnable() { @Override public void run() { showStartReward(call); } });
          return;
        }
        if (UNITY.equals(prov) && uReady && uRewardOk) {
          act.runOnUiThread(new Runnable() { @Override public void run() { showUnityReward(act, call); } });
          return;
        }
      }
    }
    reloadRewards();
    JSObject r = new JSObject(); r.put("completed", false); r.put("error", "not_ready"); call.resolve(r);
  }

  private void showStartReward(final PluginCall call) {
    sRewardOk = false; sRewardEarned = false;
    final boolean[] done = { false };
    sReward.showAd(new AdDisplayListener() {
      @Override public void adDisplayed(Ad a) { }
      @Override public void adClicked(Ad a) { }
      @Override public void adHidden(Ad a) { if (done[0]) return; done[0] = true; boolean ok = sRewardEarned; reloadRewards(); resolveReward(call, ok, SIO, null); }
      @Override public void adNotDisplayed(Ad a) { if (done[0]) return; done[0] = true; reloadRewards(); resolveReward(call, false, SIO, "show_failed"); }
    });
  }

  private void showUnityReward(Activity act, final PluginCall call) {
    uRewardOk = false;
    UnityAds.show(act, uReward, new UnityAdsShowOptions(), new IUnityAdsShowListener() {
      @Override public void onUnityAdsShowFailure(String id, UnityAds.UnityAdsShowError e, String m) { reloadRewards(); resolveReward(call, false, UNITY, "show_failed"); }
      @Override public void onUnityAdsShowStart(String id) { }
      @Override public void onUnityAdsShowClick(String id) { }
      @Override public void onUnityAdsShowComplete(String id, UnityAds.UnityAdsShowCompletionState s) {
        reloadRewards();
        resolveReward(call, s == UnityAds.UnityAdsShowCompletionState.COMPLETED, UNITY, null);
      }
    });
  }

  private void resolveReward(PluginCall call, boolean completed, String provider, String error) {
    JSObject r = new JSObject(); r.put("completed", completed); r.put("provider", provider);
    if (error != null) r.put("error", error);
    call.resolve(r);
  }

  // ------------------------------------------------------------------ banner (rodapé, acima da barra do sistema)
  @PluginMethod
  public void showBanner(final PluginCall call) {
    final Activity act = getActivity();
    if (act == null || !(sReady || uReady)) { call.reject("ads_not_ready"); return; }
    act.runOnUiThread(new Runnable() {
      @Override public void run() {
        removeBanner();
        tryBanner(act, order(), 0);
        JSObject r = new JSObject(); r.put("heightDp", BANNER_H);
        call.resolve(r);
      }
    });
  }

  /** Tenta o banner da primeira rede; se ela falhar em carregar, passa para a próxima. */
  private void tryBanner(final Activity act, final String[] ord, final int idx) {
    if (idx >= ord.length) return;
    final String prov = ord[idx];
    final ViewGroup content = act.findViewById(android.R.id.content);
    float d = act.getResources().getDisplayMetrics().density;
    FrameLayout.LayoutParams lp;
    final View view;

    if (SIO.equals(prov) && sReady) {
      Banner b = new Banner(act);
      b.setVisibility(View.GONE);                // só aparece quando carregar (evita faixa vazia)
      b.setBannerListener(new BannerListener() {
        @Override public void onReceiveAd(View v) { v.setVisibility(View.VISIBLE); }
        @Override public void onFailedToReceiveAd(View v) { nextBanner(act, ord, idx, v); }
        @Override public void onImpression(View v) { }
        @Override public void onClick(View v) { }
      });
      view = b;
      lp = new FrameLayout.LayoutParams(ViewGroup.LayoutParams.WRAP_CONTENT, ViewGroup.LayoutParams.WRAP_CONTENT, Gravity.BOTTOM | Gravity.CENTER_HORIZONTAL);
    } else if (UNITY.equals(prov) && uReady && uBanner != null) {
      BannerView b = new BannerView(act, uBanner, new UnityBannerSize(BANNER_W, BANNER_H));
      b.setVisibility(View.GONE);
      b.setListener(new BannerView.IListener() {
        @Override public void onBannerLoaded(BannerView v) { v.setVisibility(View.VISIBLE); }
        @Override public void onBannerShown(BannerView v) { }
        @Override public void onBannerClick(BannerView v) { }
        @Override public void onBannerFailedToLoad(BannerView v, BannerErrorInfo info) { nextBanner(act, ord, idx, v); }
        @Override public void onBannerLeftApplication(BannerView v) { }
      });
      view = b;
      lp = new FrameLayout.LayoutParams((int) (BANNER_W * d), (int) (BANNER_H * d), Gravity.BOTTOM | Gravity.CENTER_HORIZONTAL);
    } else {
      tryBanner(act, ord, idx + 1);
      return;
    }

    WindowInsetsCompat wi = ViewCompat.getRootWindowInsets(content);
    if (wi != null) { Insets bars = wi.getInsets(WindowInsetsCompat.Type.systemBars()); lp.bottomMargin = bars.bottom; }
    content.addView(view, lp);
    banner = view;
    if (view instanceof BannerView) ((BannerView) view).load();
  }

  private void nextBanner(final Activity act, final String[] ord, final int idx, final View failed) {
    act.runOnUiThread(new Runnable() {
      @Override public void run() {
        if (banner != failed) return;            // já saiu da tela
        removeBanner();
        tryBanner(act, ord, idx + 1);
      }
    });
  }

  @PluginMethod
  public void hideBanner(final PluginCall call) {
    final Activity act = getActivity();
    if (act == null) { call.resolve(); return; }
    act.runOnUiThread(new Runnable() { @Override public void run() { removeBanner(); call.resolve(); } });
  }

  private void removeBanner() {
    if (banner == null) return;
    try {
      ViewGroup p = (ViewGroup) banner.getParent();
      if (p != null) p.removeView(banner);
      if (banner instanceof BannerView) ((BannerView) banner).destroy();
    } catch (Exception ignored) { }
    banner = null;
  }

  @Override
  protected void handleOnDestroy() { removeBanner(); super.handleOnDestroy(); }
}
