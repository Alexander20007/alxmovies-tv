package __PKG__;

import android.content.ActivityNotFoundException;
import android.content.ComponentName;
import android.content.Intent;
import android.content.pm.PackageInfo;
import android.content.pm.PackageManager;
import android.graphics.Bitmap;
import android.graphics.Canvas;
import android.graphics.drawable.BitmapDrawable;
import android.graphics.drawable.Drawable;
import android.net.Uri;
import android.util.Base64;
import com.getcapacitor.JSArray;
import com.getcapacitor.JSObject;
import com.getcapacitor.Plugin;
import com.getcapacitor.PluginCall;
import com.getcapacitor.PluginMethod;
import com.getcapacitor.annotation.CapacitorPlugin;
import java.io.ByteArrayOutputStream;

/**
 * Entrega o link do vídeo direto para a tela de "adicionar download" do 1DM / ADM (sem abrir o navegador deles),
 * já com o nome do arquivo preenchido. Segue a integração oficial do 1DM (Util1DM): ACTION_VIEW + componente
 * idm.internet.download.manager.Downloader + extras extra_filename / extra_referer / extra_useragent.
 */
@CapacitorPlugin(name = "AlxDownload")
public class AlxDownloadPlugin extends Plugin {
  private static final String[] IDS = { "adm", "1dm" };
  private static final String[] NAMES = { "ADM", "1DM" };
  private static final String[][] PKGS = {
    { "com.dv.adm", "com.dv.adm.pay" },
    { "idm.internet.download.manager.plus", "idm.internet.download.manager", "idm.internet.download.manager.adm.lite" },
  };
  private static final String IDM_ACTIVITY = "idm.internet.download.manager.Downloader";
  private static final String ADM_ACTIVITY = "com.dv.adm.AEditor";

  private String installedPkg(int idx) {
    PackageManager pm = getContext().getPackageManager();
    for (String p : PKGS[idx]) {
      try { pm.getPackageInfo(p, 0); return p; } catch (Exception ignored) {}
    }
    return null;
  }

  /** Ícone do app instalado (PNG em base64) para mostrar na lista, como no Android. */
  private String iconOf(String pkg) {
    try {
      Drawable d = getContext().getPackageManager().getApplicationIcon(pkg);
      int size = 96;
      Bitmap bmp;
      if (d instanceof BitmapDrawable && ((BitmapDrawable) d).getBitmap() != null) {
        bmp = Bitmap.createScaledBitmap(((BitmapDrawable) d).getBitmap(), size, size, true);
      } else {
        bmp = Bitmap.createBitmap(size, size, Bitmap.Config.ARGB_8888);
        Canvas c = new Canvas(bmp);
        d.setBounds(0, 0, size, size);
        d.draw(c);
      }
      ByteArrayOutputStream bo = new ByteArrayOutputStream();
      bmp.compress(Bitmap.CompressFormat.PNG, 100, bo);
      return "data:image/png;base64," + Base64.encodeToString(bo.toByteArray(), Base64.NO_WRAP);
    } catch (Throwable t) {
      return "";
    }
  }

  @PluginMethod
  public void listManagers(PluginCall call) {
    JSArray arr = new JSArray();
    for (int i = 0; i < IDS.length; i++) {
      String pkg = installedPkg(i);
      JSObject o = new JSObject();
      o.put("id", IDS[i]);
      o.put("name", NAMES[i]);
      o.put("installed", pkg != null);
      o.put("icon", pkg == null ? "" : iconOf(pkg));
      arr.put(o);
    }
    JSObject r = new JSObject();
    r.put("managers", arr);
    call.resolve(r);
  }

  @PluginMethod
  public void open(PluginCall call) {
    String id = call.getString("id", "other");
    String url = call.getString("url");
    if (url == null || !(url.startsWith("https://") || url.startsWith("http://"))) { call.reject("Link inválido."); return; }
    String filename = call.getString("filename");
    if (filename == null || filename.isEmpty()) filename = call.getString("title");
    String referer = call.getString("referer");
    String ua = call.getString("userAgent");

    try {
      if ("1dm".equals(id)) {
        String pkg = installedPkg(1);
        if (pkg == null) { call.reject("not_installed"); return; }
        // Componente explícito = abre a janela "Baixar arquivo" do 1DM (sem passar pelo navegador dele)
        Intent i = new Intent(Intent.ACTION_VIEW);
        i.setComponent(new ComponentName(pkg, IDM_ACTIVITY));
        i.setData(Uri.parse(url));
        i.putExtra("secure_uri", true); // esconde o link do vídeo dentro do 1DM (versões antigas ignoram)
        if (filename != null && !filename.isEmpty()) i.putExtra("extra_filename", filename);
        if (referer != null && !referer.isEmpty()) i.putExtra("extra_referer", referer);
        if (ua != null && !ua.isEmpty()) i.putExtra("extra_useragent", ua);
        i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
        getContext().startActivity(i);
      } else if ("adm".equals(id)) {
        String pkg = installedPkg(0);
        if (pkg == null) { call.reject("not_installed"); return; }
        Intent i = new Intent(Intent.ACTION_VIEW);
        i.setComponent(new ComponentName(pkg, ADM_ACTIVITY));
        i.setData(Uri.parse(url));
        i.putExtra(Intent.EXTRA_TEXT, url);
        if (filename != null && !filename.isEmpty()) i.putExtra("com.android.extra.filename", filename);
        i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
        try {
          getContext().startActivity(i);
        } catch (ActivityNotFoundException e) {
          // versão do ADM com outro nome de tela: usa o "compartilhar" do próprio app
          Intent s = new Intent(Intent.ACTION_SEND);
          s.setType("text/plain");
          s.setPackage(pkg);
          s.putExtra(Intent.EXTRA_TEXT, url);
          s.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
          getContext().startActivity(s);
        }
      } else {
        Intent i = new Intent(Intent.ACTION_VIEW, Uri.parse(url));
        if (filename != null && !filename.isEmpty()) { i.putExtra("extra_filename", filename); i.putExtra(Intent.EXTRA_TITLE, filename); }
        if (referer != null && !referer.isEmpty()) i.putExtra("extra_referer", referer);
        Intent ch = Intent.createChooser(i, "Baixar com…");
        ch.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
        getContext().startActivity(ch);
      }
      call.resolve();
    } catch (ActivityNotFoundException e) {
      call.reject("not_installed");
    } catch (Exception e) {
      call.reject(e.getMessage() == null ? "Não foi possível abrir o gerenciador." : e.getMessage());
    }
  }

  @PluginMethod
  public void openStore(PluginCall call) {
    String id = call.getString("id", "adm");
    int idx = "adm".equals(id) ? 0 : 1;
    String pkgName = PKGS[idx][idx == 0 ? 0 : 1]; // ADM: com.dv.adm — 1DM: idm.internet.download.manager
    try {
      Intent i = new Intent(Intent.ACTION_VIEW, Uri.parse("market://details?id=" + pkgName));
      i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
      getContext().startActivity(i);
    } catch (Exception e) {
      Intent i = new Intent(Intent.ACTION_VIEW, Uri.parse("https://play.google.com/store/apps/details?id=" + pkgName));
      i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
      getContext().startActivity(i);
    }
    call.resolve();
  }

  /** Abre as configurações de espelhamento de tela do Android (Transmitir / Smart View). */
  @PluginMethod
  public void openCast(PluginCall call) {
    String[] actions = { "android.settings.CAST_SETTINGS", "android.settings.WIFI_DISPLAY_SETTINGS", android.provider.Settings.ACTION_SETTINGS };
    for (String a : actions) {
      try {
        Intent i = new Intent(a);
        i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
        getContext().startActivity(i);
        call.resolve();
        return;
      } catch (Exception ignored) {}
    }
    call.reject("Não foi possível abrir o espelhamento.");
  }
}
