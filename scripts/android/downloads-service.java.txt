package __PKG__;

import android.app.Notification;
import android.app.NotificationChannel;
import android.app.NotificationManager;
import android.app.PendingIntent;
import android.app.Service;
import android.content.Context;
import android.content.Intent;
import android.content.pm.ServiceInfo;
import android.os.Build;
import android.os.IBinder;
import androidx.core.app.NotificationCompat;
import androidx.core.content.ContextCompat;

/** Mantém os downloads vivos com o app fechado e mostra a notificação com o progresso. Só liga enquanto há download ativo. */
public class AlxDownloadService extends Service {
  static final String CH = "alx_downloads";
  static final int ID = 7720;
  private static volatile boolean running = false;

  /** Mostra ou atualiza a notificação (liga o serviço se ainda não estiver ligado). */
  public static void show(Context c, int active, int queued, int pct, String title) {
    try {
      Notification n = build(c, active, queued, pct, title);
      if (running) {
        NotificationManager nm = (NotificationManager) c.getSystemService(Context.NOTIFICATION_SERVICE);
        if (nm != null) nm.notify(ID, n);
        return;
      }
      Intent i = new Intent(c, AlxDownloadService.class);
      i.putExtra("active", active); i.putExtra("queued", queued); i.putExtra("pct", pct); i.putExtra("title", title);
      ContextCompat.startForegroundService(c, i);
    } catch (Throwable ignored) { /* sem permissão ou app em segundo plano: o download continua enquanto o processo viver */ }
  }

  public static void hide(Context c) {
    try { if (running) c.stopService(new Intent(c, AlxDownloadService.class)); } catch (Throwable ignored) { }
  }

  private static Notification build(Context c, int active, int queued, int pct, String title) {
    NotificationManager nm = (NotificationManager) c.getSystemService(Context.NOTIFICATION_SERVICE);
    if (Build.VERSION.SDK_INT >= 26 && nm != null && nm.getNotificationChannel(CH) == null) {
      NotificationChannel ch = new NotificationChannel(CH, "Downloads", NotificationManager.IMPORTANCE_LOW);
      ch.setDescription("Progresso dos vídeos que estão sendo baixados");
      nm.createNotificationChannel(ch);
    }
    Intent open = c.getPackageManager().getLaunchIntentForPackage(c.getPackageName());
    PendingIntent pi = open == null ? null : PendingIntent.getActivity(c, 0, open, PendingIntent.FLAG_IMMUTABLE | PendingIntent.FLAG_UPDATE_CURRENT);
    String text = active > 1 ? active + " vídeos baixando" : (title == null || title.isEmpty() ? "Baixando vídeo" : title);
    if (queued > 0) text += " · " + queued + " na fila";
    return new NotificationCompat.Builder(c, CH)
      .setSmallIcon(android.R.drawable.stat_sys_download)
      .setContentTitle("ALXmovies · Downloads")
      .setContentText(text)
      .setProgress(100, Math.max(0, Math.min(100, pct)), false)
      .setOngoing(true)
      .setOnlyAlertOnce(true)
      .setContentIntent(pi)
      .build();
  }

  @Override public int onStartCommand(Intent intent, int flags, int startId) {
    int active = intent == null ? 1 : intent.getIntExtra("active", 1);
    int queued = intent == null ? 0 : intent.getIntExtra("queued", 0);
    int pct = intent == null ? 0 : intent.getIntExtra("pct", 0);
    String title = intent == null ? "" : intent.getStringExtra("title");
    Notification n = build(this, active, queued, pct, title);
    if (Build.VERSION.SDK_INT >= 29) startForeground(ID, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_DATA_SYNC);
    else startForeground(ID, n);
    running = true;
    return START_NOT_STICKY;
  }

  @Override public void onDestroy() { running = false; super.onDestroy(); }
  @Override public IBinder onBind(Intent intent) { return null; }
}
