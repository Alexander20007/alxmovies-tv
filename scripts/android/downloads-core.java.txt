/**
 * Baixador interno do ALXmovies. Fila + trabalho em segundo plano (sobrevive ao fechar a tela).
 *  - HLS (.m3u8): escolhe a qualidade, baixa os trechos em paralelo, decifra AES-128 e guarda uma playlist LOCAL
 *    (index.m3u8) que o player nativo toca offline, com barra de tempo. Também exporta como um único arquivo .ts/.mp4.
 *  - Arquivo direto (.mp4/.ts/...): download único, com retomada.
 * Tudo fica em Android/data/<pacote>/files/downloads (sem pedir permissão; some ao desinstalar).
 */
public final class AlxDownloadCore {
  public interface Listener { void onChanged(JSONObject item); void onRemoved(String id); }

  static final String DEFAULT_UA = "Mozilla/5.0 (Linux; Android 13) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0 Mobile Safari/537.36";
  private static final long MIN_FREE_BYTES = 150L * 1024 * 1024;
  private static final int DEFAULT_THREADS = 16;   // conexões ao mesmo tempo (dividido entre os downloads ativos)
  private static final int IO_BUF = 131072;        // 128 KB por leitura/escrita

  static {
    // mais conexões reaproveitadas (keep-alive): evita refazer a conexão segura a cada trecho
    try { System.setProperty("http.maxConnections", "64"); System.setProperty("http.keepAlive", "true"); } catch (Throwable ignored) { }
  }

  // ------------------------------------------------------------------ tipos internos
  static final class CancelledEx extends RuntimeException { CancelledEx() { super("cancelado"); } }
  static final class HttpEx extends IOException { final int code; HttpEx(int c) { super("HTTP " + c); code = c; } }
  static final class UserEx extends Exception { final String errCode; UserEx(String code, String msg) { super(msg); errCode = code; } }

  static final class Run { volatile boolean cancel = false; }

  static final class Item {
    String id, title = "", subtitle = "", poster = "", meta = "", url = "", referer = "", ua = "";
    int maxHeight = 0;
    volatile String status = "queued";   // queued | downloading | paused | done | error
    String kind = "";                    // hls | file
    String ext = "ts";                   // ts | mp4 | mkv | webm
    boolean exportable = true;
    volatile int segsTotal = 0;
    final AtomicInteger segsDone = new AtomicInteger();
    final AtomicLong bytes = new AtomicLong();        // recebido (inclui trechos incompletos)
    final AtomicLong doneBytes = new AtomicLong();    // soma dos trechos já completos
    volatile long sizeBytes = 0, speed = 0, totalHint = 0;
    volatile double durationSec = 0;
    volatile String error = "", errorCode = "";
    long createdAt, finishedAt;
    volatile Run run;
    long lastEmit;
  }

  static final class Seg { String url; double dur; boolean disc; String keyUri, keyIv; }
  static final class Plan {
    List<Seg> segs = new ArrayList<>(); String mapUri; long mediaSeq = 0; int target = 0; boolean endList = false;
    double total() { double t = 0; for (Seg s : segs) t += s.dur; return t; }
  }
  static final class Task { String url; File file; String keyUri, keyIv; long seq; }
  static final class Prepared {
    List<Task> tasks = new ArrayList<>();
    final Map<String, byte[]> keys = new HashMap<>();
    final AtomicInteger maxParallel = new AtomicInteger(8);   // cai sozinho se o servidor reclamar de muitas conexões
    final AtomicInteger inFlight = new AtomicInteger(0);
  }
  static final class Fetched { String text; URL url; }
  static final class Variant { String url; long bw; int h; String audio; }
  static final class Media { String group, uri; boolean def; }

  // ------------------------------------------------------------------ estado
  private static AlxDownloadCore inst;
  public static synchronized AlxDownloadCore get(Context c) {
    if (inst == null) inst = new AlxDownloadCore(c.getApplicationContext());
    return inst;
  }

  private final Context app;
  private final File root, dbFile;
  private final LinkedHashMap<String, Item> items = new LinkedHashMap<>();
  private final ExecutorService runner = Executors.newCachedThreadPool();
  private volatile Listener listener;
  private volatile boolean wifiOnly = false;
  private volatile int maxActive = 2;
  private volatile int segThreads = DEFAULT_THREADS;
  private boolean started = false;
  private long lastService = 0;

  private AlxDownloadCore(Context c) {
    app = c;
    File ext = c.getExternalFilesDir("downloads");
    root = ext != null ? ext : new File(c.getFilesDir(), "downloads");
    root.mkdirs();
    dbFile = new File(c.getFilesDir(), "alx_downloads.json");
    load();
  }

  public void setListener(Listener l) { listener = l; }
  public boolean isWifiOnly() { return wifiOnly; }

  public void setOptions(boolean wifi, int max, int threads) {
    wifiOnly = wifi;
    maxActive = Math.max(1, Math.min(3, max));
    segThreads = Math.max(2, Math.min(32, threads));
    kick();
  }

  /** Chamado quando o app abre: retoma o que estava baixando quando o app foi fechado. */
  public void ensureStarted() {
    synchronized (items) {
      if (started) return;
      started = true;
    }
    kick();
  }

  // ------------------------------------------------------------------ persistência
  private void load() {
    try {
      if (!dbFile.exists()) return;
      FileInputStream in = new FileInputStream(dbFile);
      ByteArrayOutputStream bo = new ByteArrayOutputStream();
      byte[] b = new byte[8192]; int n;
      while ((n = in.read(b)) > 0) bo.write(b, 0, n);
      in.close();
      JSONArray arr = new JSONArray(bo.toString("UTF-8"));
      for (int i = 0; i < arr.length(); i++) {
        JSONObject o = arr.getJSONObject(i);
        Item it = new Item();
        it.id = o.getString("id");
        it.title = o.optString("title"); it.subtitle = o.optString("subtitle"); it.poster = o.optString("poster"); it.meta = o.optString("meta");
        it.url = o.optString("url"); it.referer = o.optString("referer"); it.ua = o.optString("ua");
        it.maxHeight = o.optInt("maxHeight", 0);
        it.status = o.optString("status", "paused"); it.kind = o.optString("kind"); it.ext = o.optString("ext", "ts");
        it.exportable = o.optBoolean("exportable", true);
        it.segsTotal = o.optInt("segsTotal", 0); it.sizeBytes = o.optLong("sizeBytes", 0); it.totalHint = o.optLong("totalHint", 0);
        it.durationSec = o.optDouble("durationSec", 0);
        it.error = o.optString("error"); it.errorCode = o.optString("errorCode");
        it.createdAt = o.optLong("createdAt", System.currentTimeMillis()); it.finishedAt = o.optLong("finishedAt", 0);
        File dir = new File(root, it.id);
        if (it.status.equals("done")) {
          if (!dir.exists()) continue;                       // arquivos apagados por fora
          it.bytes.set(it.sizeBytes); it.segsDone.set(it.segsTotal);
        } else {
          if (it.status.equals("downloading")) it.status = "queued";   // o app foi fechado: volta para a fila
          recount(it, dir);
        }
        items.put(it.id, it);
      }
    } catch (Exception ignored) { /* banco corrompido: começa vazio */ }
  }

  private void save() {
    try {
      JSONArray arr = new JSONArray();
      synchronized (items) {
        for (Item it : items.values()) {
          JSONObject o = new JSONObject();
          o.put("id", it.id); o.put("title", it.title); o.put("subtitle", it.subtitle); o.put("poster", it.poster); o.put("meta", it.meta);
          o.put("url", it.url); o.put("referer", it.referer); o.put("ua", it.ua); o.put("maxHeight", it.maxHeight);
          o.put("status", it.status); o.put("kind", it.kind); o.put("ext", it.ext); o.put("exportable", it.exportable);
          o.put("segsTotal", it.segsTotal); o.put("sizeBytes", it.sizeBytes); o.put("totalHint", it.totalHint); o.put("durationSec", it.durationSec);
          o.put("error", it.error); o.put("errorCode", it.errorCode); o.put("createdAt", it.createdAt); o.put("finishedAt", it.finishedAt);
          arr.put(o);
        }
      }
      File tmp = new File(dbFile.getPath() + ".tmp");
      FileOutputStream out = new FileOutputStream(tmp);
      out.write(arr.toString().getBytes("UTF-8"));
      out.close();
      if (!tmp.renameTo(dbFile)) { dbFile.delete(); tmp.renameTo(dbFile); }
    } catch (Exception ignored) { }
  }

  private void recount(Item it, File dir) {
    int n = 0; long b = 0;
    File[] subs = dir.listFiles();
    if (subs != null) for (File f : subs) {
      if (f.isDirectory()) {
        File[] fs = f.listFiles();
        if (fs != null) for (File x : fs) if (x.isFile() && !x.getName().endsWith(".part") && x.length() > 0) { n++; b += x.length(); }
      } else if (f.isFile() && !f.getName().endsWith(".m3u8") && !f.getName().startsWith("poster.")) {
        b += f.length();
        if (!f.getName().endsWith(".part")) n++;
      }
    }
    if (it.kind.equals("file")) { it.bytes.set(b); it.doneBytes.set(b); it.segsDone.set(0); }
    else { it.segsDone.set(n); it.bytes.set(b); it.doneBytes.set(b); }
  }

  // ------------------------------------------------------------------ JSON para o JavaScript
  public JSONObject toJson(Item it) {
    JSONObject o = new JSONObject();
    try {
      boolean done = it.status.equals("done");
      long bytes = done ? it.sizeBytes : it.bytes.get();
      long est; double prog;
      if (done) { est = it.sizeBytes; prog = 1; }
      else if (it.kind.equals("file")) { est = it.totalHint; prog = it.totalHint > 0 ? Math.min(1.0, (double) bytes / it.totalHint) : 0; }
      else {
        int tot = it.segsTotal, d = it.segsDone.get();
        prog = tot > 0 ? Math.min(1.0, (double) d / tot) : 0;
        est = d > 0 && tot > 0 ? it.doneBytes.get() * tot / d : 0;
      }
      o.put("id", it.id); o.put("title", it.title); o.put("subtitle", it.subtitle); o.put("poster", it.poster); o.put("meta", it.meta);
      o.put("status", it.status); o.put("kind", it.kind); o.put("ext", it.ext); o.put("exportable", it.exportable);
      o.put("bytes", bytes); o.put("estTotal", est); o.put("progress", prog);
      o.put("segsDone", it.segsDone.get()); o.put("segsTotal", it.segsTotal);
      o.put("speed", it.status.equals("downloading") ? it.speed : 0);
      o.put("durationSec", it.durationSec);
      o.put("error", it.error); o.put("errorCode", it.errorCode);
      o.put("createdAt", it.createdAt); o.put("finishedAt", it.finishedAt);
      o.put("waitingWifi", it.status.equals("queued") && wifiOnly && isMetered());
      File pf = new File(root, it.id + "/poster.jpg");
      o.put("posterFile", pf.exists() && pf.length() > 0 ? pf.getAbsolutePath() : "");
    } catch (Exception ignored) { }
    return o;
  }

  public JSONArray list() {
    JSONArray a = new JSONArray();
    synchronized (items) {
      List<Item> l = new ArrayList<>(items.values());
      Collections.reverse(l);   // mais novos primeiro
      for (Item it : l) a.put(toJson(it));
    }
    return a;
  }

  private void emit(Item it, boolean force) {
    long now = System.currentTimeMillis();
    if (!force && now - it.lastEmit < 400) return;
    it.lastEmit = now;
    Listener l = listener;
    if (l != null) l.onChanged(toJson(it));
    syncService(force);
  }

  private void syncService(boolean force) {
    long now = System.currentTimeMillis();
    if (!force && now - lastService < 1000) return;
    lastService = now;
    int active = 0, queued = 0; double sum = 0; String title = "";
    synchronized (items) {
      for (Item it : items.values()) {
        if (it.status.equals("downloading")) {
          active++; title = it.title;
          int tot = it.segsTotal;
          sum += it.kind.equals("file") ? (it.totalHint > 0 ? (double) it.bytes.get() / it.totalHint : 0) : (tot > 0 ? (double) it.segsDone.get() / tot : 0);
        } else if (it.status.equals("queued")) queued++;
      }
    }
    if (active == 0) AlxDownloadService.hide(app);
    else AlxDownloadService.show(app, active, queued, (int) Math.round(sum / active * 100), title);
  }

  // ------------------------------------------------------------------ comandos
  public JSONObject enqueue(JSONObject o) throws Exception {
    String id = o.getString("id");
    String url = o.optString("url", "");
    if (url.isEmpty()) throw new UserEx("args", "Link do vídeo ausente");
    Item it;
    synchronized (items) {
      it = items.get(id);
      if (it != null) {
        if (it.status.equals("done")) return toJson(it);
        it.url = url;
        if (o.has("referer")) it.referer = o.optString("referer");
        if (o.has("maxHeight")) it.maxHeight = o.optInt("maxHeight", it.maxHeight);
        if (!it.status.equals("downloading")) { it.status = "queued"; it.error = ""; it.errorCode = ""; }
      } else {
        it = new Item();
        it.id = id;
        it.title = o.optString("title", "Vídeo"); it.subtitle = o.optString("subtitle"); it.poster = o.optString("poster"); it.meta = o.optString("meta");
        it.url = url; it.referer = o.optString("referer"); it.ua = o.optString("userAgent", DEFAULT_UA);
        it.maxHeight = o.optInt("maxHeight", 0);
        it.createdAt = System.currentTimeMillis();
        items.put(id, it);
      }
    }
    save();
    emit(it, true);
    kick();
    return toJson(it);
  }

  public void updateUrl(String id, String url, String referer) {
    Item it; synchronized (items) { it = items.get(id); }
    if (it == null || url == null || url.isEmpty()) return;
    it.url = url;
    if (referer != null) it.referer = referer;
    save();
  }

  public void pause(String id) {
    Item it; synchronized (items) { it = items.get(id); }
    if (it == null) return;
    if (it.status.equals("downloading") || it.status.equals("queued")) {
      Run r = it.run; if (r != null) r.cancel = true;
      it.status = "paused"; it.speed = 0;
      save(); emit(it, true); kick();
    }
  }

  public void resume(String id) {
    Item it; synchronized (items) { it = items.get(id); }
    if (it == null) return;
    if (it.status.equals("paused") || it.status.equals("error")) {
      it.status = "queued"; it.error = ""; it.errorCode = "";
      save(); emit(it, true); kick();
    }
  }

  public void pauseAll() {
    List<String> ids = new ArrayList<>();
    synchronized (items) { for (Item it : items.values()) if (it.status.equals("downloading") || it.status.equals("queued")) ids.add(it.id); }
    for (String id : ids) pause(id);
  }

  public void resumeAll() {
    List<String> ids = new ArrayList<>();
    synchronized (items) { for (Item it : items.values()) if (it.status.equals("paused") || it.status.equals("error")) ids.add(it.id); }
    for (String id : ids) resume(id);
  }

  public void remove(String id, boolean deleteFiles) {
    Item it;
    synchronized (items) { it = items.remove(id); }
    if (it == null) return;
    Run r = it.run; if (r != null) r.cancel = true;
    it.status = "paused";
    save();
    Listener l = listener; if (l != null) l.onRemoved(id);
    final File dir = new File(root, id);
    if (deleteFiles) runner.submit(new Runnable() {
      @Override public void run() {
        try { Thread.sleep(700); } catch (InterruptedException ignored) { }
        deleteRecursive(dir);
      }
    });
    syncService(true);
    kick();
  }

  /** Começa o que está na fila, respeitando o máximo de downloads ao mesmo tempo e a opção "só no Wi-Fi". */
  public void kick() {
    synchronized (items) {
      int active = 0;
      for (Item it : items.values()) if (it.status.equals("downloading")) active++;
      boolean blocked = wifiOnly && isMetered();
      for (Item it : items.values()) {
        if (active >= maxActive || blocked) break;
        if (it.status.equals("queued")) { start(it); active++; }
      }
    }
    syncService(true);
  }

  private void start(final Item it) {
    final Run run = new Run();
    it.run = run;
    it.status = "downloading"; it.error = ""; it.errorCode = ""; it.speed = 0;
    save();
    emit(it, true);
    runner.submit(new Runnable() { @Override public void run() { work(it, run); } });
  }

  private boolean isMetered() {
    try {
      ConnectivityManager cm = (ConnectivityManager) app.getSystemService(Context.CONNECTIVITY_SERVICE);
      return cm != null && cm.isActiveNetworkMetered();
    } catch (Exception e) { return false; }
  }

  // ------------------------------------------------------------------ trabalho de um item
  private void work(Item it, Run run) {
    try {
      File dir = new File(root, it.id);
      dir.mkdirs();
      if (freeBytes() < MIN_FREE_BYTES) throw new UserEx("space", "Pouco espaço livre no celular. Libere espaço e tente de novo.");
      Prepared p = prepare(it, run, dir);
      if (it.kind.equals("file")) downloadFile(it, run, dir);
      else downloadSegments(it, run, p);
      if (run.cancel) return;
      it.sizeBytes = dirSize(dir);
      it.bytes.set(it.sizeBytes);
      it.finishedAt = System.currentTimeMillis();
      it.speed = 0;
      it.status = "done";
      save();
      emit(it, true);
    } catch (CancelledEx c) {
      // pausa ou remoção: nada a fazer
    } catch (UserEx e) {
      fail(it, run, e.errCode, e.getMessage());
    } catch (HttpEx e) {
      boolean expired = e.code == 401 || e.code == 403 || e.code == 410;
      fail(it, run, expired ? "expired" : "http", expired ? "O link do vídeo expirou. Toque em Tentar de novo." : "O servidor do vídeo respondeu com erro (" + e.code + ").");
    } catch (Throwable e) {
      fail(it, run, "network", "Sem conexão ou servidor instável. Toque em Tentar de novo.");
    } finally {
      kick();
    }
  }

  private void fail(Item it, Run run, String code, String msg) {
    if (run.cancel || it.run != run) return;
    it.status = "error"; it.error = msg; it.errorCode = code; it.speed = 0;
    save(); emit(it, true);
  }

  // ------------------------------------------------------------------ playlist
  private Prepared prepare(Item it, Run run, File dir) throws Exception {
    Prepared prep = new Prepared();
    HttpURLConnection c = open(it.url, it, 0);
    int code = c.getResponseCode();
    if (code >= 400) { c.disconnect(); throw new HttpEx(code); }
    String ctype = c.getContentType() == null ? "" : c.getContentType().toLowerCase(Locale.ROOT);
    URL base = c.getURL();
    BufferedInputStream in = new BufferedInputStream(c.getInputStream(), 65536);
    in.mark(16);
    byte[] head = new byte[7];
    int got = 0, r;
    while (got < 7 && (r = in.read(head, got, 7 - got)) > 0) got += r;
    in.reset();
    boolean playlist = got >= 7 && new String(head, "UTF-8").equals("#EXTM3U");

    if (!playlist) {
      c.disconnect();
      if (ctype.contains("text/html") || ctype.contains("json")) throw new UserEx("format", "O link não é um vídeo.");
      it.kind = "file";
      it.ext = guessExt(ctype, base.getPath());
      it.exportable = true;
      savePoster(it, dir);
      return prep;
    }

    String text = readAll(in, 4 * 1024 * 1024);
    c.disconnect();
    it.kind = "hls";

    Plan v; Plan a = null;
    String audioUrl = null; long bw = 2000000;
    if (text.contains("#EXT-X-STREAM-INF")) {
      List<Variant> vs = new ArrayList<>(); List<Media> ms = new ArrayList<>();
      String[] lines = text.split("\\r?\\n");
      for (int i = 0; i < lines.length; i++) {
        String ln = lines[i].trim();
        if (ln.startsWith("#EXT-X-STREAM-INF:")) {
          Map<String, String> at = attrs(ln.substring(18));
          String next = null;
          for (int j = i + 1; j < lines.length; j++) { String t = lines[j].trim(); if (!t.isEmpty() && !t.startsWith("#")) { next = t; break; } }
          if (next == null) continue;
          Variant va = new Variant();
          va.url = resolve(base, next);
          try { va.bw = Long.parseLong(at.containsKey("AVERAGE-BANDWIDTH") ? at.get("AVERAGE-BANDWIDTH") : at.get("BANDWIDTH")); } catch (Exception e) { va.bw = 0; }
          String res = at.get("RESOLUTION");
          if (res != null && res.contains("x")) { try { va.h = Integer.parseInt(res.substring(res.indexOf('x') + 1).trim()); } catch (Exception ignored) { } }
          va.audio = at.get("AUDIO");
          vs.add(va);
        } else if (ln.startsWith("#EXT-X-MEDIA:")) {
          Map<String, String> at = attrs(ln.substring(13));
          if ("AUDIO".equalsIgnoreCase(at.get("TYPE")) && at.get("URI") != null) {
            Media m = new Media(); m.group = at.get("GROUP-ID"); m.uri = resolve(base, at.get("URI")); m.def = "YES".equalsIgnoreCase(at.get("DEFAULT"));
            ms.add(m);
          }
        }
      }
      if (vs.isEmpty()) throw new UserEx("format", "Não encontramos vídeo nessa lista.");
      Variant best = null;
      if (it.maxHeight > 0) {
        for (Variant x : vs) if (x.h > 0 && x.h <= it.maxHeight && (best == null || x.h > best.h || (x.h == best.h && x.bw > best.bw))) best = x;
        if (best == null) for (Variant x : vs) if (best == null || x.h < best.h || (x.h == best.h && x.bw < best.bw)) best = x;
      } else {
        for (Variant x : vs) if (best == null || x.bw > best.bw || (x.bw == best.bw && x.h > best.h)) best = x;
      }
      bw = best.bw > 0 ? best.bw : 2000000;
      if (best.audio != null) {
        Media pick = null;
        for (Media m : ms) if (best.audio.equals(m.group) && (pick == null || (m.def && !pick.def))) pick = m;
        if (pick != null) audioUrl = pick.uri;
      }
      Fetched fv = fetchText(best.url, it);
      v = parseMedia(fv.text, fv.url);
      if (audioUrl != null) { Fetched fa = fetchText(audioUrl, it); a = parseMedia(fa.text, fa.url); }
    } else {
      v = parseMedia(text, base);
    }
    if (run.cancel) throw new CancelledEx();

    // arquivos que já existem só valem se a lista tem o mesmo tamanho de antes
    int total = v.segs.size() + (v.mapUri != null ? 1 : 0) + (a == null ? 0 : a.segs.size() + (a.mapUri != null ? 1 : 0));
    if (it.segsTotal != 0 && it.segsTotal != total) { deleteRecursive(new File(dir, "v")); deleteRecursive(new File(dir, "a")); }
    it.segsTotal = total;

    addTasks(prep, v, new File(dir, "v"));
    if (a != null) addTasks(prep, a, new File(dir, "a"));

    int done = 0; long bytes = 0;
    for (Task t : prep.tasks) if (t.file.exists() && t.file.length() > 0) { done++; bytes += t.file.length(); }
    it.segsDone.set(done); it.bytes.set(bytes); it.doneBytes.set(bytes);

    it.ext = v.mapUri != null ? "mp4" : "ts";
    it.exportable = a == null;
    it.durationSec = v.total();

    // playlists locais (o player toca estas, sem internet)
    String segExt = v.mapUri != null ? ".m4s" : ".ts";
    if (a == null) {
      writeText(new File(dir, "index.m3u8"), mediaText(v, "v", segExt));
    } else {
      String aExt = a.mapUri != null ? ".m4s" : ".ts";
      writeText(new File(dir, "video.m3u8"), mediaText(v, "v", segExt));
      writeText(new File(dir, "audio.m3u8"), mediaText(a, "a", aExt));
      writeText(new File(dir, "index.m3u8"),
        "#EXTM3U\n#EXT-X-MEDIA:TYPE=AUDIO,GROUP-ID=\"aud\",NAME=\"Audio\",DEFAULT=YES,AUTOSELECT=YES,URI=\"audio.m3u8\"\n"
        + "#EXT-X-STREAM-INF:BANDWIDTH=" + bw + ",AUDIO=\"aud\"\nvideo.m3u8\n");
    }
    savePoster(it, dir);
    save();
    emit(it, true);
    return prep;
  }

  /** Guarda a capa no celular para a lista de downloads aparecer completa mesmo sem internet. */
  private void savePoster(Item it, File dir) {
    try {
      File f = new File(dir, "poster.jpg");
      if (f.exists() && f.length() > 0) return;
      if (it.poster == null || !it.poster.startsWith("http")) return;
      HttpURLConnection c = open(it.poster, it, 0);
      try {
        if (c.getResponseCode() >= 400) return;
        InputStream in = c.getInputStream();
        File tmp = new File(dir, "poster.tmp");
        FileOutputStream o = new FileOutputStream(tmp);
        try {
          byte[] b = new byte[8192]; int n; long tot = 0;
          while ((n = in.read(b)) > 0) { o.write(b, 0, n); tot += n; if (tot > 3000000) throw new IOException("capa grande demais"); }
        } finally { o.close(); in.close(); }
        if (!tmp.renameTo(f)) tmp.delete();
      } finally { c.disconnect(); }
    } catch (Exception ignored) { /* capa é opcional */ }
  }

  private void addTasks(Prepared prep, Plan p, File sub) {
    sub.mkdirs();
    String ext = p.mapUri != null ? ".m4s" : ".ts";
    if (p.mapUri != null) { Task t = new Task(); t.url = p.mapUri; t.file = new File(sub, "init.mp4"); prep.tasks.add(t); }
    for (int i = 0; i < p.segs.size(); i++) {
      Seg s = p.segs.get(i);
      Task t = new Task();
      t.url = s.url; t.keyUri = s.keyUri; t.keyIv = s.keyIv; t.seq = p.mediaSeq + i;
      t.file = new File(sub, String.format(Locale.US, "seg_%05d%s", i, ext));
      prep.tasks.add(t);
    }
  }

  private String mediaText(Plan p, String sub, String segExt) {
    double maxDur = 0; for (Seg s : p.segs) maxDur = Math.max(maxDur, s.dur);
    int target = Math.max(p.target, (int) Math.ceil(maxDur));
    StringBuilder sb = new StringBuilder();
    sb.append("#EXTM3U\n#EXT-X-VERSION:").append(p.mapUri != null ? 7 : 3).append("\n");
    sb.append("#EXT-X-TARGETDURATION:").append(Math.max(target, 1)).append("\n#EXT-X-MEDIA-SEQUENCE:0\n#EXT-X-PLAYLIST-TYPE:VOD\n");
    if (p.mapUri != null) sb.append("#EXT-X-MAP:URI=\"").append(sub).append("/init.mp4\"\n");
    for (int i = 0; i < p.segs.size(); i++) {
      Seg s = p.segs.get(i);
      if (s.disc) sb.append("#EXT-X-DISCONTINUITY\n");
      sb.append("#EXTINF:").append(String.format(Locale.US, "%.3f", s.dur)).append(",\n");
      sb.append(sub).append("/").append(String.format(Locale.US, "seg_%05d%s", i, segExt)).append("\n");
    }
    sb.append("#EXT-X-ENDLIST\n");
    return sb.toString();
  }

  private Plan parseMedia(String text, URL base) throws Exception {
    Plan p = new Plan();
    String keyUri = null, keyIv = null;
    boolean disc = false;
    double dur = 0;
    for (String raw : text.split("\\r?\\n")) {
      String ln = raw.trim();
      if (ln.isEmpty()) continue;
      if (ln.startsWith("#EXT-X-MEDIA-SEQUENCE:")) { try { p.mediaSeq = Long.parseLong(ln.substring(22).trim()); } catch (Exception ignored) { } }
      else if (ln.startsWith("#EXT-X-TARGETDURATION:")) { try { p.target = (int) Math.ceil(Double.parseDouble(ln.substring(22).trim())); } catch (Exception ignored) { } }
      else if (ln.startsWith("#EXT-X-ENDLIST")) p.endList = true;
      else if (ln.startsWith("#EXT-X-KEY:")) {
        Map<String, String> at = attrs(ln.substring(11));
        String m = at.get("METHOD") == null ? "NONE" : at.get("METHOD").toUpperCase(Locale.ROOT);
        if (m.equals("AES-128")) { keyUri = at.get("URI") == null ? null : resolve(base, at.get("URI")); keyIv = at.get("IV"); }
        else if (m.equals("NONE")) { keyUri = null; keyIv = null; }
        else throw new UserEx("drm", "Este vídeo é protegido e não pode ser baixado.");
      }
      else if (ln.startsWith("#EXT-X-MAP:")) {
        Map<String, String> at = attrs(ln.substring(11));
        if (at.get("BYTERANGE") != null) throw new UserEx("format", "Formato de vídeo não suportado para download.");
        if (at.get("URI") != null && p.mapUri == null) p.mapUri = resolve(base, at.get("URI"));
      }
      else if (ln.equals("#EXT-X-DISCONTINUITY")) disc = true;
      else if (ln.startsWith("#EXT-X-BYTERANGE")) throw new UserEx("format", "Formato de vídeo não suportado para download.");
      else if (ln.startsWith("#EXTINF:")) {
        String d = ln.substring(8); int i = d.indexOf(',');
        if (i >= 0) d = d.substring(0, i);
        try { dur = Double.parseDouble(d.trim()); } catch (Exception e) { dur = 0; }
      }
      else if (!ln.startsWith("#")) {
        Seg s = new Seg();
        s.url = resolve(base, ln); s.dur = dur; s.disc = disc; s.keyUri = keyUri; s.keyIv = keyIv;
        p.segs.add(s);
        disc = false; dur = 0;
      }
    }
    if (!p.endList) throw new UserEx("live", "Esta é uma transmissão ao vivo; não dá para baixar.");
    if (p.segs.isEmpty()) throw new UserEx("format", "A lista do vídeo veio vazia.");
    return p;
  }

  private static Map<String, String> attrs(String s) {
    Map<String, String> m = new HashMap<>();
    int i = 0, n = s.length();
    while (i < n) {
      int eq = s.indexOf('=', i);
      if (eq < 0) break;
      String k = s.substring(i, eq).trim().toUpperCase(Locale.ROOT);
      int j = eq + 1;
      String v;
      if (j < n && s.charAt(j) == '"') { int e = s.indexOf('"', j + 1); if (e < 0) e = n; v = s.substring(j + 1, e); i = e + 2; }
      else { int e = s.indexOf(',', j); if (e < 0) e = n; v = s.substring(j, e).trim(); i = e + 1; }
      m.put(k, v);
    }
    return m;
  }

  private static String resolve(URL base, String ref) throws IOException {
    return new URL(base, ref.trim()).toString();
  }

  // ------------------------------------------------------------------ rede
  private HttpURLConnection open(String u, Item it, long rangeStart) throws IOException { return open(u, it, rangeStart, false); }

  private HttpURLConnection open(String u, Item it, long rangeStart, boolean identity) throws IOException {
    String cur = u;
    for (int i = 0; i < 6; i++) {
      HttpURLConnection c = (HttpURLConnection) new URL(cur).openConnection();
      c.setInstanceFollowRedirects(false);
      c.setConnectTimeout(15000);
      c.setReadTimeout(25000);
      c.setRequestProperty("User-Agent", it.ua == null || it.ua.isEmpty() ? DEFAULT_UA : it.ua);
      c.setRequestProperty("Accept", "*/*");
      if (identity) c.setRequestProperty("Accept-Encoding", "identity");   // vídeo já é comprimido: sem gzip, sem custo à toa
      if (it.referer != null && !it.referer.isEmpty()) {
        c.setRequestProperty("Referer", it.referer);
        try { URL r = new URL(it.referer); c.setRequestProperty("Origin", r.getProtocol() + "://" + r.getHost()); } catch (Exception ignored) { }
      }
      if (rangeStart > 0) c.setRequestProperty("Range", "bytes=" + rangeStart + "-");
      int code = c.getResponseCode();
      if (code == 301 || code == 302 || code == 303 || code == 307 || code == 308) {
        String loc = c.getHeaderField("Location");
        c.disconnect();
        if (loc == null) throw new HttpEx(code);
        cur = new URL(new URL(cur), loc).toString();
        continue;
      }
      return c;
    }
    throw new IOException("redirecionamentos demais");
  }

  private Fetched fetchText(String u, Item it) throws IOException {
    HttpURLConnection c = open(u, it, 0);
    try {
      int code = c.getResponseCode();
      if (code >= 400) throw new HttpEx(code);
      Fetched f = new Fetched();
      f.url = c.getURL();
      f.text = readAll(new BufferedInputStream(c.getInputStream(), 65536), 4 * 1024 * 1024);
      return f;
    } finally { c.disconnect(); }
  }

  private static String readAll(InputStream in, int max) throws IOException {
    ByteArrayOutputStream bo = new ByteArrayOutputStream();
    byte[] b = new byte[16384]; int n;
    while ((n = in.read(b)) > 0) { bo.write(b, 0, n); if (bo.size() > max) throw new IOException("lista grande demais"); }
    return bo.toString("UTF-8");
  }

  // ------------------------------------------------------------------ baixar trechos (HLS)
  private int activeCount() {
    int n = 0;
    synchronized (items) { for (Item x : items.values()) if (x.status.equals("downloading")) n++; }
    return Math.max(1, n);
  }

  private static void throttleDown(Prepared p) {
    int v;
    do { v = p.maxParallel.get(); } while (v > 2 && !p.maxParallel.compareAndSet(v, Math.max(2, v - 2)));
  }

  private void downloadSegments(final Item it, final Run run, final Prepared p) throws Exception {
    final int threads = Math.max(2, Math.min(32, segThreads / activeCount()));   // 16 conexões num download; 8 + 8 em dois
    p.maxParallel.set(threads);
    final ExecutorService pool = Executors.newFixedThreadPool(threads);
    ExecutorCompletionService<Boolean> ecs = new ExecutorCompletionService<>(pool);
    int submitted = 0;
    try {
      for (final Task t : p.tasks) {
        if (t.file.exists() && t.file.length() > 0) continue;
        ecs.submit(new java.util.concurrent.Callable<Boolean>() {
          @Override public Boolean call() throws Exception { fetchTask(it, run, p, t); return Boolean.TRUE; }
        });
        submitted++;
      }
      int finished = 0;
      long lastBytes = it.bytes.get(), lastT = System.currentTimeMillis();
      while (finished < submitted) {
        if (run.cancel) throw new CancelledEx();
        Future<Boolean> f = ecs.poll(400, TimeUnit.MILLISECONDS);
        if (f != null) {
          try { f.get(); }
          catch (ExecutionException e) { Throwable c = e.getCause(); if (c instanceof Exception) throw (Exception) c; throw e; }
          finished++;
        }
        long now = System.currentTimeMillis();
        if (now - lastT >= 1000) {
          long b = it.bytes.get();
          long inst = Math.max(0, (b - lastBytes) * 1000 / (now - lastT));
          it.speed = it.speed == 0 ? inst : (it.speed * 6 + inst * 4) / 10;
          lastBytes = b; lastT = now;
        }
        if (wifiOnly && isMetered()) { waitWifi(it, run); throw new CancelledEx(); }
        emit(it, false);
      }
    } finally {
      pool.shutdownNow();
    }
  }

  private void waitWifi(Item it, Run run) {
    run.cancel = true;
    it.status = "queued"; it.speed = 0;
    save(); emit(it, true);
  }

  /** Espera a vez (o limite de conexões cai sozinho se o servidor reclamar) e baixa um trecho. */
  private void fetchTask(Item it, Run run, Prepared p, Task t) throws Exception {
    while (true) {
      if (run.cancel) throw new CancelledEx();
      int cur = p.inFlight.get();
      if (cur < p.maxParallel.get() && p.inFlight.compareAndSet(cur, cur + 1)) break;
      Thread.sleep(25);
    }
    try { fetchOne(it, run, p, t); }
    finally { p.inFlight.decrementAndGet(); }
  }

  private void fetchOne(Item it, Run run, Prepared p, Task t) throws Exception {
    byte[] key = null, iv = null;
    if (t.keyUri != null) { key = getKey(it, p, t.keyUri); iv = ivFor(t); }
    Exception last = null;
    for (int attempt = 0; attempt < 6; attempt++) {
      if (run.cancel) throw new CancelledEx();
      File part = new File(t.file.getPath() + ".part");
      long counted = 0;
      HttpURLConnection c = null;
      boolean ok = false;
      try {
        c = open(t.url, it, 0, true);
        int code = c.getResponseCode();
        if (code >= 400) throw new HttpEx(code);
        InputStream in = new BufferedInputStream(c.getInputStream(), IO_BUF);
        Cipher ci = null;
        if (key != null) {
          ci = Cipher.getInstance("AES/CBC/PKCS5Padding");
          ci.init(Cipher.DECRYPT_MODE, new SecretKeySpec(key, "AES"), new IvParameterSpec(iv));
        }
        FileOutputStream out = new FileOutputStream(part);
        try {
          byte[] buf = new byte[IO_BUF]; int n;
          while ((n = in.read(buf)) > 0) {
            if (run.cancel) throw new CancelledEx();
            if (ci != null) {
              byte[] o = ci.update(buf, 0, n);
              if (o != null && o.length > 0) { out.write(o); counted += o.length; it.bytes.addAndGet(o.length); }
            } else {
              out.write(buf, 0, n); counted += n; it.bytes.addAndGet(n);
            }
          }
          if (ci != null) {
            byte[] o = ci.doFinal();
            if (o != null && o.length > 0) { out.write(o); counted += o.length; it.bytes.addAndGet(o.length); }
          }
        } finally { try { out.close(); } catch (IOException ignored) { } try { in.close(); } catch (IOException ignored) { } }
        if (counted == 0) throw new IOException("trecho vazio");
        if (!part.renameTo(t.file)) throw new IOException("não foi possível salvar o trecho");
        it.doneBytes.addAndGet(counted);
        it.segsDone.incrementAndGet();
        ok = true;
        return;                                  // conexão devolvida ao "pool": o próximo trecho reaproveita (sem novo handshake)
      } catch (CancelledEx e) {
        it.bytes.addAndGet(-counted); part.delete(); throw e;
      } catch (HttpEx e) {
        it.bytes.addAndGet(-counted); part.delete();
        if (e.code == 401 || e.code == 403 || e.code == 410) throw e;
        if (e.code == 429 || e.code == 503) throttleDown(p);   // servidor reclamou de muitas conexões: reduz o ritmo
        last = e;
      } catch (IOException e) {
        it.bytes.addAndGet(-counted); part.delete(); last = e;
      } catch (GeneralSecurityException e) {
        it.bytes.addAndGet(-counted); part.delete(); last = e;
      } finally {
        if (c != null && !ok) c.disconnect();    // só fecha a conexão se deu erro
      }
      long wait = 600L * (attempt + 1);
      for (long w = 0; w < wait; w += 100) { if (run.cancel) throw new CancelledEx(); Thread.sleep(100); }
    }
    if (last instanceof HttpEx) throw last;
    throw new IOException("Falha de rede ao baixar um trecho do vídeo");
  }

  private byte[] getKey(Item it, Prepared p, String uri) throws Exception {
    synchronized (p.keys) {
      byte[] k = p.keys.get(uri);
      if (k != null) return k;
      HttpURLConnection c = open(uri, it, 0);
      try {
        int code = c.getResponseCode();
        if (code >= 400) throw new HttpEx(code);
        ByteArrayOutputStream bo = new ByteArrayOutputStream();
        InputStream in = c.getInputStream();
        byte[] b = new byte[256]; int n;
        while ((n = in.read(b)) > 0 && bo.size() < 4096) bo.write(b, 0, n);
        k = bo.toByteArray();
      } finally { c.disconnect(); }
      if (k.length != 16) throw new UserEx("drm", "Este vídeo é protegido e não pode ser baixado.");
      p.keys.put(uri, k);
      return k;
    }
  }

  private static byte[] ivFor(Task t) {
    byte[] iv = new byte[16];
    if (t.keyIv != null && !t.keyIv.isEmpty()) {
      String h = t.keyIv.startsWith("0x") || t.keyIv.startsWith("0X") ? t.keyIv.substring(2) : t.keyIv;
      while (h.length() < 32) h = "0" + h;
      if (h.length() > 32) h = h.substring(h.length() - 32);
      for (int i = 0; i < 16; i++) iv[i] = (byte) Integer.parseInt(h.substring(i * 2, i * 2 + 2), 16);
    } else {
      long s = t.seq;
      for (int i = 15; i >= 8; i--) { iv[i] = (byte) (s & 0xff); s >>= 8; }
    }
    return iv;
  }

  // ------------------------------------------------------------------ arquivo direto
  private void downloadFile(Item it, Run run, File dir) throws Exception {
    File out = new File(dir, "video." + it.ext);
    File part = new File(dir, "video.part");
    if (out.exists() && out.length() > 0) { return; }
    long have = part.exists() ? part.length() : 0;
    HttpURLConnection c = open(it.url, it, have);
    try {
      int code = c.getResponseCode();
      if (code == 416 && have > 0) { part.renameTo(out); return; }
      if (code >= 400) throw new HttpEx(code);
      boolean append = code == 206 && have > 0;
      if (!append) have = 0;
      long len = c.getContentLength() > 0 ? c.getContentLength() : 0;
      String cr = c.getHeaderField("Content-Range");
      if (cr != null && cr.contains("/")) { try { len = Long.parseLong(cr.substring(cr.lastIndexOf('/') + 1).trim()) - have; } catch (Exception ignored) { } }
      it.totalHint = len > 0 ? len + have : 0;
      it.bytes.set(have); it.doneBytes.set(have);
      InputStream in = new BufferedInputStream(c.getInputStream(), IO_BUF);
      FileOutputStream os = new FileOutputStream(part, append);
      long lastBytes = have, lastT = System.currentTimeMillis();
      try {
        byte[] buf = new byte[IO_BUF]; int n;
        while ((n = in.read(buf)) > 0) {
          if (run.cancel) throw new CancelledEx();
          os.write(buf, 0, n);
          it.bytes.addAndGet(n);
          long now = System.currentTimeMillis();
          if (now - lastT >= 1000) {
            long b = it.bytes.get();
            long inst = Math.max(0, (b - lastBytes) * 1000 / (now - lastT));
            it.speed = it.speed == 0 ? inst : (it.speed * 6 + inst * 4) / 10;
            lastBytes = b; lastT = now;
            if (wifiOnly && isMetered()) { waitWifi(it, run); throw new CancelledEx(); }
          }
          emit(it, false);
        }
      } finally { try { os.close(); } catch (IOException ignored) { } try { in.close(); } catch (IOException ignored) { } }
    } finally { c.disconnect(); }
    if (!part.renameTo(out)) throw new IOException("não foi possível salvar o arquivo");
  }

  private static String guessExt(String ctype, String path) {
    String p = path == null ? "" : path.toLowerCase(Locale.ROOT);
    if (ctype.contains("mp2t") || p.endsWith(".ts")) return "ts";
    if (ctype.contains("matroska") || p.endsWith(".mkv")) return "mkv";
    if (ctype.contains("webm") || p.endsWith(".webm")) return "webm";
    return "mp4";
  }

  /** Endereço local para o player nativo (file://...). Null se o download ainda não terminou. */
  public String localUrl(String id) {
    Item it; synchronized (items) { it = items.get(id); }
    if (it == null || !it.status.equals("done")) return null;
    File f = it.kind.equals("file") ? new File(root, id + "/video." + it.ext) : new File(root, id + "/index.m3u8");

  }

  /** Junta tudo em um único arquivo .ts/.mp4 na pasta Downloads/ALXmovies do celular. */
  public String export(String id) throws Exception {
    Item it; synchronized (items) { it = items.get(id); }
    if (it == null || !it.status.equals("done")) throw new UserEx("state", "O download ainda não terminou.");
    if (!it.exportable) throw new UserEx("state", "Este vídeo tem áudio separado e não pode virar um arquivo único.");
    if (Build.VERSION.SDK_INT < 29) throw new UserEx("state", "Salvar na pasta Downloads exige Android 10 ou mais novo.");
    File dir = new File(root, it.id);
    List<File> parts = new ArrayList<>();
    if (it.kind.equals("file")) parts.add(new File(dir, "video." + it.ext));
    else {
      File init = new File(dir, "v/init.mp4");
      if (init.exists()) parts.add(init);
      File[] segs = new File(dir, "v").listFiles();
      if (segs != null) {
        Arrays.sort(segs);
        for (File s : segs) if (s.getName().startsWith("seg_") && !s.getName().endsWith(".part")) parts.add(s);
      }
    }
    if (parts.isEmpty()) throw new UserEx("state", "Arquivos do download não encontrados.");
    String base = it.title.replaceAll("[\\\\/:*?\"<>|]+", " ").replaceAll("\\s+", " ").trim();
    if (base.isEmpty()) base = "ALXmovies";
    if (base.length() > 100) base = base.substring(0, 100);
    String name = base + "." + it.ext;
    String mime = it.ext.equals("ts") ? "video/mp2t" : it.ext.equals("mkv") ? "video/x-matroska" : it.ext.equals("webm") ? "video/webm" : "video/mp4";

    ContentResolver res = app.getContentResolver();
    ContentValues cv = new ContentValues();
    cv.put(MediaStore.MediaColumns.DISPLAY_NAME, name);
    cv.put(MediaStore.MediaColumns.MIME_TYPE, mime);
    cv.put(MediaStore.MediaColumns.RELATIVE_PATH, Environment.DIRECTORY_DOWNLOADS + "/ALXmovies");
    cv.put(MediaStore.MediaColumns.IS_PENDING, 1);
    Uri uri = res.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, cv);
    if (uri == null) throw new UserEx("state", "Não foi possível criar o arquivo em Downloads.");
    boolean ok = false;
    try {
      OutputStream os = res.openOutputStream(uri);
      if (os == null) throw new IOException("sem saída");
      try {
        byte[] buf = new byte[65536];
        for (File f : parts) {
          FileInputStream in = new FileInputStream(f);
          try { int n; while ((n = in.read(buf)) > 0) os.write(buf, 0, n); } finally { in.close(); }
        }
      } finally { os.close(); }
      ContentValues done = new ContentValues();
      done.put(MediaStore.MediaColumns.IS_PENDING, 0);
      res.update(uri, done, null, null);
      ok = true;
    } finally {
      if (!ok) { try { res.delete(uri, null, null); } catch (Exception ignored) { } }
    }
    return "Download/ALXmovies/" + name;
  }

  // ------------------------------------------------------------------ utilidades de arquivo
  static long dirSize(File f) {
    if (f.isFile()) return f.length();
    long s = 0;
    File[] fs = f.listFiles();
    if (fs != null) for (File x : fs) s += dirSize(x);
    return s;
  }

  static void deleteRecursive(File f) {
    if (f == null || !f.exists()) return;
    if (f.isDirectory()) { File[] fs = f.listFiles(); if (fs != null) for (File x : fs) deleteRecursive(x); }
    f.delete();
  }

  private static void writeText(File f, String s) throws IOException {
    FileOutputStream o = new FileOutputStream(f);
    try { o.write(s.getBytes("UTF-8")); } finally { o.close(); }
  }
}
