package __PKG__;

import com.getcapacitor.JSObject;
import com.getcapacitor.Plugin;
import com.getcapacitor.PluginCall;
import com.getcapacitor.PluginMethod;
import com.getcapacitor.annotation.CapacitorPlugin;
import org.json.JSONObject;

/** Ponte JavaScript ⇄ baixador interno (AlxDownloadCore). */
@CapacitorPlugin(name = "AlxDownloads")
public class AlxDownloadsPlugin extends Plugin implements AlxDownloadCore.Listener {
  private AlxDownloadCore core;

  @Override public void load() {
    core = AlxDownloadCore.get(getContext());
    core.setListener(this);
  }

  @Override public void onChanged(JSONObject item) {
    JSObject d = new JSObject(); d.put("item", item);
    notifyListeners("changed", d);
  }
  @Override public void onRemoved(String id) {
    JSObject d = new JSObject(); d.put("id", id);
    notifyListeners("removed", d);
  }

  @PluginMethod public void list(PluginCall call) {
    core.ensureStarted();
    JSObject r = new JSObject();
    r.put("items", core.list());
    r.put("wifiOnly", core.isWifiOnly());
    call.resolve(r);
  }

  @PluginMethod public void enqueue(PluginCall call) {
    try {
      JSObject r = new JSObject();
      r.put("item", core.enqueue(call.getData()));
      call.resolve(r);
    } catch (Exception e) { call.reject(e.getMessage() == null ? "erro" : e.getMessage()); }
  }

  @PluginMethod public void pause(PluginCall call) { core.pause(call.getString("id", "")); call.resolve(); }
  @PluginMethod public void resume(PluginCall call) { core.resume(call.getString("id", "")); call.resolve(); }
  @PluginMethod public void pauseAll(PluginCall call) { core.pauseAll(); call.resolve(); }
  @PluginMethod public void resumeAll(PluginCall call) { core.resumeAll(); call.resolve(); }
  @PluginMethod public void kick(PluginCall call) { core.kick(); call.resolve(); }

  @PluginMethod public void remove(PluginCall call) {
    core.remove(call.getString("id", ""), Boolean.TRUE.equals(call.getBoolean("deleteFiles", true)));
    call.resolve();
  }

  @PluginMethod public void updateUrl(PluginCall call) {
    core.updateUrl(call.getString("id", ""), call.getString("url", ""), call.getString("referer"));
    call.resolve();
  }

  @PluginMethod public void setOptions(PluginCall call) {
    core.setOptions(Boolean.TRUE.equals(call.getBoolean("wifiOnly", false)), call.getInt("maxActive", 2), call.getInt("threads", 16));
    call.resolve();
  }

  @PluginMethod public void storage(PluginCall call) {
    JSObject r = new JSObject();
    JSONObject s = core.storage();
    r.put("downloadsBytes", s.optLong("downloadsBytes"));
    r.put("freeBytes", s.optLong("freeBytes"));
    r.put("totalBytes", s.optLong("totalBytes"));
    call.resolve(r);
  }

  @PluginMethod public void playUrl(PluginCall call) {
    String u = core.localUrl(call.getString("id", ""));
    if (u == null) { call.reject("not_ready"); return; }
    JSObject r = new JSObject(); r.put("url", u);
    call.resolve(r);
  }

  /** Junta os trechos num único arquivo e salva em Downloads/ALXmovies (pode demorar em vídeos grandes). */
  @PluginMethod public void exportFile(final PluginCall call) {
    final String id = call.getString("id", "");
    new Thread(new Runnable() {
      @Override public void run() {
        try {
          String where = core.export(id);
          JSObject r = new JSObject(); r.put("path", where);
          call.resolve(r);
        } catch (Exception e) { call.reject(e.getMessage() == null ? "Não foi possível exportar." : e.getMessage()); }
      }
    }, "alx-export").start();
  }
}
